import datetime,hashlib,json,os,pathlib,subprocess,sys,time
label,cwd,*cmd=sys.argv[1:]
root=pathlib.Path('/tmp/nofx-audit-repairs-20260913')
def git(*args):return subprocess.check_output(['git',*args],cwd=root,text=True).strip()
rev=git('rev-parse','HEAD');dirty=git('status','--porcelain')
if dirty:raise SystemExit('Refusing final verification on dirty source')
out=pathlib.Path('/tmp/nofx-final-verification');log=out/(label+'.log')
env=os.environ.copy();env['GIT_DIR']='/tmp/structural-daily-loss-release/nofx/.git/worktrees/nofx-audit-repairs-20260913';env['GIT_WORK_TREE']=str(root)
start=datetime.datetime.now(datetime.timezone.utc).isoformat();t=time.monotonic()
with log.open('wb') as f:r=subprocess.run(cmd,cwd=cwd,env=env,stdout=f,stderr=subprocess.STDOUT)
end=git('rev-parse','HEAD');dirty_after=git('status','--porcelain')
record={'task':label,'command':cmd,'cwd':cwd,'source_revision':rev,'end_revision':end,'source_clean_before':True,'source_clean_after':not dirty_after,'started_utc':start,'duration_seconds':round(time.monotonic()-t,3),'exit_code':r.returncode,'log':str(log),'log_bytes':log.stat().st_size,'log_sha256':hashlib.sha256(log.read_bytes()).hexdigest(),'revision_unchanged':rev==end}
(out/(label+'.json')).write_text(json.dumps(record,indent=2)+'\n');print(json.dumps(record),flush=True)
sys.exit(r.returncode or (10 if rev!=end or dirty_after else 0))
