## 1. The 21 big-move windows, re-derived from bars (NOT from their tape.csv)
sqlite3 -header -csv "file:/home/hoang/nofx/data/data.db?mode=ro" "
WITH b AS (SELECT open_time_ms,o,c,(open_time_ms/1000/900)*900 AS w FROM bars
           WHERE symbol='MNQ' AND tf='1m'
             AND open_time_ms >= strftime('%s','2026-09-02 05:00:00')*1000
             AND open_time_ms <  strftime('%s','2026-09-04 05:00:00')*1000),
agg AS (SELECT w, datetime(w,'unixepoch','-5 hours') AS win_ct, COUNT(*) n,
        (SELECT o FROM b b2 WHERE b2.w=b.w ORDER BY b2.open_time_ms ASC  LIMIT 1) AS wopen,
        (SELECT c FROM b b3 WHERE b3.w=b.w ORDER BY b3.open_time_ms DESC LIMIT 1) AS wclose
        FROM b GROUP BY w)
SELECT win_ct,n,wopen,wclose,ROUND(wclose-wopen,2) FROM agg WHERE ABS(wclose-wopen)>=40 ORDER BY w;"
#   -> exactly 21 rows, identical to their list. Denominator CONFIRMED.
