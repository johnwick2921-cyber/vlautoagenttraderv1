func confirmAcceptanceRule(rule string) string {
	switch rule {
	case "15m_close": // legacy: stored docs only
		return "15m-close"
	case "1x5m_close":
		return "5m-close" // the A2-fixed one-close rule
	default:
		return "2x5m" // "2x5m_close"
	}
}

func baselineFirstConfirmFireMs(c kernel.PlanConfirm, bars []market.Kline, sinceMs, nowMs int64) int64 {
	if c.Rule == "touch" {
		return 0
	}
	w := kernel.BarsSince(bars, sinceMs)
	if len(w) == 0 {
		return 0
	}
	// E5 — 1m_mss fires at the breaking bar.
	if c.Rule == "1m_mss" {
		m := kernel.EvaluateMSS(w, c.Side, nowMs)
		if m.Met {
			return m.BreakTimeMs
		}
		return 0
	}
	// E6 — time_hold fires when the qualifying run completes (its first bar).
	if c.Rule == "time_hold" {
		need := kernel.AcceptHoldMin()
		above := strings.EqualFold(c.Side, "above")
		run := 0
		for i, b := range w {
			if b.CloseTime >= nowMs {
				continue
			}
			beyond := (above && b.Close > c.RefPrice) || (!above && b.Close < c.RefPrice)
			if beyond {
				run++
				if run >= need {
					return w[i-run+1].OpenTime
				}
			} else {
				run = 0
			}
		}
		return 0
	}
	rule := confirmAcceptanceRule(c.Rule)
	var dur, need int64
	switch rule {
	case "5m-close":
		dur, need = 5*60_000, 1
	case "2x5m":
		dur, need = 5*60_000, 2
	default:
		return 0
	}
	buckets := kernel.AggregateBars(w, dur)
	above := strings.EqualFold(c.Side, "above")
	run := int64(0)
	for _, b := range buckets {
		if b.CloseTime > nowMs {
			continue
		}
		beyond := (above && b.Close > c.RefPrice) || (!above && b.Close < c.RefPrice)
		if beyond {
			run++
			if run >= need {
				return b.OpenTime
			}
		} else {
			run = 0
		}
	}
	return 0
}
